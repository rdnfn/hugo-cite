{{- /* This shortcode renders the full citation of one work in place, in the same style as the {{< bibliography >}} entries (authors, year, title, venue, DOI or URL). Use it for a further reading list, one work per line: - {{< fullcite "Lessig 2002" >}} - {{< fullcite "Nussbaum2011" >}} The key must match an ID in the CSL-JSON bibliography file. The work is not added to the cited works, so it is not listed by {{< bibliography cited >}}. */ -}} {{- $key := .Get 0 | default (.Get "key") -}} {{- if not $key -}} {{- errorf "fullcite: missing citation key. Example: {{< fullcite \"Faure 1909\" >}} (%s)" .Position -}} {{- end -}} {{- $citationStyle := "apa" -}} {{- with site.Params.citationStyle -}} {{- if templates.Exists (printf "partials/bibliography/%s-style.html" .) -}} {{- $citationStyle = . -}} {{- end -}} {{- end -}} {{- $bib := dict -}} {{- with resources.Get "bibliography.json" -}} {{- $bib = . | transform.Unmarshal -}} {{- else -}} {{- errorf "Unable to get resource %q" "bibliography.json" -}} {{- end -}} {{- range first 1 (where $bib "id" "eq" $key) -}} {{ partial (printf "bibliography/%s-style.html" $citationStyle) . | strings.TrimSpace | safeHTML }} {{- else -}} No matching key was found for `{{ $key }}` in the references. Please make sure to provide an available ID in your `bib.json` file. {{- end -}}
